-- ------------------------------------------------------------
--  TRIGGERS untuk SIM Ditolak & Expired
-- ------------------------------------------------------------

DROP TRIGGER IF EXISTS trg_pengguna_validasi_sim;

DELIMITER $$

CREATE TRIGGER trg_pengguna_validasi_sim
AFTER INSERT
ON pengguna
FOR EACH ROW
BEGIN

    -- SIM sudah kadaluarsa
    IF NEW.tanggal_kadaluarsa_sim < CURDATE() THEN

        UPDATE pengguna
        SET status_verifikasi = 'expired'
        WHERE id_pengguna = NEW.id_pengguna;

    END IF;

    -- SIM lebih dari 5 tahun dari hari ini
    IF NEW.tanggal_kadaluarsa_sim > DATE_ADD(CURDATE(), INTERVAL 5 YEAR) THEN

        UPDATE pengguna
        SET status_verifikasi = 'ditolak'
        WHERE id_pengguna = NEW.id_pengguna;

    END IF;

END$$

DELIMITER ;


-- ----------------------------------------------------------------------------------
--  TRIGGERS after insert penyewaan : ubah status kendaraan
-- ----------------------------------------------------------------------------------

DELIMITER $$

CREATE TRIGGER after_insert_penyewaan
AFTER INSERT
ON penyewaan
FOR EACH ROW
BEGIN

    UPDATE kendaraan
    SET status_kendaraan = 'disewa'
    WHERE id_kendaraan = NEW.id_kendaraan;

END$$

DELIMITER ;

-- ----------------------------------------------------------------------------------
--  TRIGGERS after insert pengembalian : ubah status penyewaan & kendaraan
-- ----------------------------------------------------------------------------------

DROP TRIGGER IF EXISTS ubah_status_setelah_pengembalian;

DELIMITER $$

CREATE TRIGGER ubah_status_setelah_pengembalian
AFTER INSERT
ON pengembalian
FOR EACH ROW

BEGIN

    UPDATE penyewaan
    SET status_penyewaan = 'selesai'
    WHERE id_penyewaan = NEW.id_penyewaan;

    UPDATE kendaraan
    SET status_kendaraan = 'tersedia'
    WHERE id_kendaraan = (

        SELECT id_kendaraan
        FROM penyewaan
        WHERE id_penyewaan = NEW.id_penyewaan

    );

END$$

DELIMITER ;

-- ----------------------------------------------------------------------------------
--  TRIGGERS after insert pengembalian : hitung denda keterlambatanh
-- ----------------------------------------------------------------------------------

DROP TRIGGER IF EXISTS denda_keterlambatan;

DELIMITER $$

CREATE TRIGGER denda_keterlambatan
AFTER INSERT
ON pengembalian
FOR EACH ROW

BEGIN

    DECLARE tanggal_rencana DATE;
    DECLARE telat INT;
    DECLARE karyawan_input INT;

    SELECT
        waktu_selesai_rencana
    INTO
        tanggal_rencana
    FROM penyewaan
    WHERE id_penyewaan = NEW.id_penyewaan;

    IF NEW.waktu_pengembalian > tanggal_rencana THEN

        SET telat =
            DATEDIFF(
                NEW.waktu_pengembalian,
                tanggal_rencana
            );

        SET karyawan_input =
            NEW.id_karyawan;

        INSERT INTO denda
        (
            id_pengembalian,
            id_karyawan,
            alasan_denda,
            nominal_denda,
            keterangan
        )
        VALUES
        (
            NEW.id_pengembalian,
            karyawan_input,
            'keterlambatan_pengembalian',
            telat * 100000,
            '-'
        );

    END IF;

END$$

DELIMITER ;
